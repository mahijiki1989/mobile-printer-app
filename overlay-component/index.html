How to Use — Minimal Template

    
<!-- 1. Include the stylesheet -->
<link rel="stylesheet" href="overlay.css">

<!-- 2. Wrap your image in .img-card -->
<div class="img-card">

  <img src="your-photo.jpg" alt="...">

  <!-- 3. Place the overlay inside .img-card -->
  <div class="geo-overlay">
    <div class="geo-row">
      <span class="geo-lbl">Latitude</span>
      <input class="geo-val" type="text"
        data-field="latitude"
        data-original="24.032752"
        value="24.032752">
    </div>
    <!-- repeat .geo-row for each field … -->
    <div class="geo-row">
      <span class="geo-lbl">Note</span>
      <textarea class="geo-val geo-val--note"
        data-field="note"
        rows="1">Your note here</textarea>
    </div>
  </div>

  <!-- Optional badge -->
  <div class="geo-badge">Powered by NoteCam</div>

</div>

<!-- 4. Include the script -->
<script src="overlay.js"></script>

<!-- 5. Listen for changes (optional) -->
<script>
  document.querySelector('.geo-overlay')
    .addEventListener('geo:change', e => {
      console.log(e.detail); // { field, value, all }
    });
</script>
